2. CONFIGURATION GOOGLE MAPS (API Key)
Dans app.json :
{
  "expo": {
    "plugins": [
      [
        "react-native-maps",
        {
          "androidGoogleMapsApiKey": "VOTRE_CLE_GOOGLE_MAPS",
          "iosGoogleMapsApiKey": "VOTRE_CLE_GOOGLE_MAPS"
        }
      ]
    ],
    "extra": {
      "googleMapsApiKey": "VOTRE_CLE_GOOGLE_MAPS"
    }
  }
}
